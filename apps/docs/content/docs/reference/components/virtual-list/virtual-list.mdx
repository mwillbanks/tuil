## API and events

Render large collections by projecting only the terminal-visible range plus configurable overscan.

```tsx
export function VirtualList<TItem>({
  items,
  renderItem,
  getItemKey,
  getItemLabel,
  height = 10,
  overscan = 1,
  offset,
  defaultOffset = 0,
  onOffsetChange,
  activeIndex,
  defaultActiveIndex = 0,
  onActiveIndexChange,
  onSelect,
  emptyMessage = "No items",
  staticLimit = 1_000,
  autoFocus,
  slots,
  slotProps,
  disabled = false,
  readOnly = false,
  ...props
}: VirtualListProps<TItem>): ReactNode;
```
